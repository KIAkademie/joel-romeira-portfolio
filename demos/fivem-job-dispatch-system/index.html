Lua NUI-Callback (dispatch.lua)

RegisterNUICallback("acceptDispatch", function(data, cb)
    SetNewWaypoint(data.x, data.y)
    TriggerServerEvent("jnr_dispatch:updateStatus", data.id, "Angenommen")
    cb("ok")
end)
