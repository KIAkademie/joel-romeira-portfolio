Geöffnete Schlösser in Folge: 0

FiveM Lua-Einbindung (client.lua)

So wird das NUI-Minispiel im FiveM-Client aufgerufen und ausgewertet:

-- FiveM Client Script: Lockpick NUI Minigame
local isMinigameOpen = false

function StartLockpickMinigame(pins, cb)
    if isMinigameOpen then return end
    isMinigameOpen = true
    SetNuiFocus(true, true)
    SendNUIMessage({
        action = "startLockpick",
        pins = pins or 4
    })
end

RegisterNUICallback("lockpickResult", function(data, cb)
    SetNuiFocus(false, false)
    isMinigameOpen = false
    if data.success then
        print("Schloss geöffnet!")
    else
        print("Dietrich abgebrochen!")
    end
    cb("ok")
end)
